% FIR filter using vector instructions.
%
% Memory layout (per the Lab 3 spec):
%   Data[0..3]     = N (input array size, multiple of 8, N <= 64)
%   Data[4..35]    = 8 weight values w[0..7]
%   Data[36 .. ]   = input array, N words
%   Output stored starting at address 0x100 (256), N-8 words.
%
% for(k = 0; k < N-8; k++) {
%     sum = 0;
%     for(i = 0; i < 8; i++)
%         sum = sum + input[k+i] * weight[i];
%     output[k] = sum;
% }
%
% The inner multiply-accumulate loop is replaced by one vector load +
% one vector multiply (v2 = v1 * v0). There is no vector "horizontal
% reduce" instruction in this ISA, so the 8 products are spilled to a
% scratch memory slot and summed with plain scalar adds.

% NOTE: with these fixed addresses (input starting at 0, output fixed
% at 0x100=256 by the spec), the input array's last byte is at
% 36 + 4*N - 1. For N > 48, that exceeds 256, so the input and output
% regions overlap and output writes can clobber not-yet-read input
% samples. This is an artifact of the address map given in the spec,
% not a bug in this program - verified correct for N up to 48
% (multiples of 8); N in {56, 64} will show corruption near the end
% of the output for this reason.

x1 = [0]         % x1 = N
x2 = 4            % address of the 8 weights
v0 = [x2]          % load the weight vector once, outside the loop

x3 = 36              % base address of the input array
x4 = 255               % output base address = 0x100 = 256, built via
x4 = x4 + 1              %   two constants since a literal is capped at 255

x13 = 255                 % scratch address (for the horizontal sum) = 480,
x13 = x13 + 225              %   built the same way; safely past the largest
                               %   possible output region (N<=64 -> <=480)

x5 = x1 - 8                     % x6 = N - 8 = number of output samples
x6 = 0                            % k = 0

.outer
x15 = x6 - x5
BGE .outer_done       % if k >= N-8, done

x16 = x6 * 4            % byte offset of input[k]
x17 = x3 + x16            % address of input[k]

v1 = [x17]                 % load input[k .. k+7]
v2 = v1 * v0                 % elementwise multiply by the weights
[x13] = v2                     % spill the 8 products for the scalar reduction

x18 = 0                          % running sum
x19 = 0                            % byte offset within the scratch slot

.inner
x15 = x19 - 32
BEQ .inner_done          % summed all 8 products?
x20 = x13 + x19
x21 = [x20]
x18 = x18 + x21
x19 = x19 + 4
BAL .inner

.inner_done
x22 = x4 + x16              % output[k] address
[x22] = x18                   % store the FIR output sample

x6 = x6 + 1
BAL .outer

.outer_done
